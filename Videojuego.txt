import random

def adivina_el_numero():
    print("¡Hola! Bienvenido al juego de adivinar el número.")
    print("Estoy pensando en un número del 1 al 10...")
    
    # Genera un número aleatorio entre 1 y 10
    numero_secreto = random.randint(1, 10)
    intentos = 3
    
    while intentos > 0:
        print(f"\nTienes {intentos} intento(s).")
        
        # Atrapa errores por si el usuario escribe letras en lugar de números xd
        try:
            tu_numero = int(input("Ingresa tu número: "))
        except ValueError:
            print("¡Eso no es un número válido! Intenta de nuevo.")
            continue
            
        # Comprueba si el usuario adivinó
        if tu_numero == numero_secreto:
            print(f"¡Felicidades! Adivinaste, el número era {numero_secreto} 🎉")
            break
        elif tu_numero < numero_secreto:
            print("Fallaste... el número es más ALTO.")
        else:
            print("Fallaste... el número es más BAJO.")
            
        intentos -= 1
        
    # Mensaje si se queda sin intentos
    if intentos == 0:
        print(f"\n¡Game Over! Te quedaste sin intentos. El número secreto era {numero_secreto}. F en el chat xd")

# Esta línea hace que el juego comience
if __name__ == "__main__":
    adivina_el_numero()
